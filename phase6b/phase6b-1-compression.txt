# Fresh world. An iron 3x3 with Compression: iron blocks, ingots and nuggets share one slot.
# 10 blocks + 5 ingots + 3 nuggets = 858 nuggets worth.
command "fill 2 4 4 4 6 6 colossalchests2:chest_wall_iron hollow"
command "setblock 3 5 4 colossalchests2:chest_core_iron"
wait --ticks 10
eval "server.execute { def be = server.overworld().getBlockEntity(dev.pos(3, 5, 4)); be.getUpgrades().setItem(0, dev.item('colossalchests2:upgrade_compression')); def s = be.getStorage(); s.insert(dev.item('minecraft:iron_block'), 10, false); s.insert(dev.item('minecraft:iron_ingot'), 5, false); s.insert(dev.item('minecraft:iron_nugget'), 3, false); s.insert(dev.item('minecraft:cobblestone'), 100, false) }"
teleport 3.5 4 1 --yaw 0 --pitch -5
use 2 5 4 --face north
wait --ticks 10
mouse-move 400,10
screenshot --name phase6b-compressed-slot
mouse-move 141,69
wait --ticks 3
screenshot --name phase6b-form-picker-blocks
# Scroll down: ingots. Shift-click is sent as its click packet, since shift can't be faked.
scroll --at 141,69 --dy -1
wait --ticks 5
screenshot --name phase6b-form-picker-ingots
eval "def cl = screen.getClass().getClassLoader(); def P = cl.loadClass('org.cyclops.colossalchests2.network.packet.ServerboundChestClickPacket'); def A = cl.loadClass('org.cyclops.colossalchests2.inventory.ChestClickAction'); cl.loadClass('org.cyclops.colossalchests2.ColossalChestsInstance').MOD.getPacketHandlerCommon().sendToServer(P.newInstance(screen.menu.containerId, 0, A.MOVE_STACK))"
wait --ticks 5
screenshot --name phase6b-after-ingots
inventory
