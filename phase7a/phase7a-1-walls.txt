# Fresh world. A wood 3x3 with two Interfaces and a Void interface on its front.
command "fill 2 4 4 4 6 6 colossalchests2:chest_wall_wood hollow"
command "setblock 2 5 4 colossalchests2:chest_wall_interface"
command "setblock 4 5 4 colossalchests2:chest_wall_interface"
command "setblock 3 6 4 colossalchests2:chest_wall_void"
teleport 3.5 4 1 --yaw 0 --pitch -10
wait --ticks 5
screenshot --name phase7a-unformed
command "setblock 3 5 4 colossalchests2:chest_core_wood"
wait --ticks 20
# Functional walls are hidden on the giant chest, like the core, until the player sneaks.
screenshot --name phase7a-formed
eval "mc.options.keyShift.setDown(true)"
wait --ticks 10
eval "player.isCrouching()"
screenshot --name phase7a-formed-sneaking
eval "mc.options.keyShift.setDown(false)"
wait --ticks 10
command "give @p minecraft:stone 16"
command "give @p minecraft:iron_nugget 4"
hold 2
# A click on an Interface opens the chest, like any wall.
use 4 5 4 --face north
wait --ticks 10
snapshot
close-screen
# Sneaking opens its settings.
eval "mc.options.keyShift.setDown(true)"
wait --ticks 5
use 4 5 4 --face north
wait --ticks 10
eval "mc.options.keyShift.setDown(false)"
snapshot
screenshot --name phase7a-settings-empty
