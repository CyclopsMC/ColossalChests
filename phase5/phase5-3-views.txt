# Fresh world. Upgrade item tooltip, the settings view with lock buttons, and an 81 slot copper chest.
command "fill 2 4 4 5 7 7 colossalchests2:chest_wall_copper hollow"
command "setblock 3 5 4 colossalchests2:chest_core_copper"
command "give @p colossalchests2:upgrade_depth"
wait --ticks 10
eval "server.execute { def be = server.overworld().getBlockEntity(dev.pos(3, 5, 4)); be.getUpgrades().setItem(0, dev.item('colossalchests2:upgrade_lock')); def s = be.getStorage(); s.insert(dev.item('minecraft:stone'), 300, false); s.insert(dev.item('minecraft:diamond'), 3, false) }"
teleport 3.5 4 1 --yaw 0 --pitch -5
use 2 5 4 --face north
wait --ticks 10
mouse-move 141,195
wait --ticks 3
screenshot --name phase5-upgrade-item-tooltip
key GLFW_KEY_LEFT_SHIFT --action press
wait --ticks 3
screenshot --name phase5-upgrade-item-tooltip-shift
key GLFW_KEY_LEFT_SHIFT --action release
click --widget "Settings"
wait --ticks 3
mouse-move 400,10
screenshot --name phase5-settings-locks
close-screen
eval "server.execute { def be = server.overworld().getBlockEntity(dev.pos(3, 5, 4)); be.getUpgrades().setItem(0, dev.item('colossalchests2:upgrade_slot_expansion')); be.getUpgrades().setItem(1, dev.item('colossalchests2:upgrade_slot_expansion')) }"
wait --ticks 5
use 2 5 4 --face north
wait --ticks 10
mouse-move 400,10
screenshot --name phase5-81-slots
